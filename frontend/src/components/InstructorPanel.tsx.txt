/** 
 * File: InstructorPanel.tsx
 * Purpose: Simple instructor management UI (local only).
 */

import React, { useState } from 'react'

/**
 * Interface: Instructor
 * Purpose: Simple instructor model for local management.
 */
interface Instructor {
  id: string
  name: string
  bio?: string
}

/**
 * Component: InstructorPanel
 * Purpose: Manage instructors in a local list (for demo).
 */
const InstructorPanel: React.FC = () => {
  const [instructors, setInstructors] = useState<Instructor[]>([
    { id: 'i-1', name: 'Jane Doe', bio: 'React instructor' },
    { id: 'i-2', name: 'John Smith', bio: 'TypeScript expert' },
  ])
  const [name, setName] = useState('')

  const add = () => {
    if (!name.trim()) return
    setInstructors(prev => [...prev, { id: `i-${Date.now()}`, name: name.trim() }])
    setName('')
  }

  const remove = (id: string) => {
    setInstructors(prev => prev.filter(i => i.id !== id))
  }

  return (
    <div className="bg-white rounded shadow-sm p-4">
      <h3 className="font-semibold mb-2">Instructors</h3>
      <ul className="space-y-2">
        {instructors.map(i => (
          <li key={i.id} className="flex items-center justify-between">
            <div>
              <div className="font-medium">{i.name}</div>
              <div className="text-xs text-slate-500">{i.bio}</div>
            </div>
            <button onClick={() => remove(i.id)} className="text-sm text-red-600">
              Remove
            </button>
          </li>
        ))}
      </ul>
      <div className="mt-3 flex gap-2">
        <input value={name} onChange={e=>setName(e.target.value)} placeholder="New instructor" className="flex-1 border rounded px-2 py-1" />
        <button onClick={add} className="px-3 py-1 bg-indigo-600 text-white rounded">Add</button>
      </div>
    </div>
  )
}

export default InstructorPanel
