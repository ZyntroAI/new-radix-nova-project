# 🚨 Incident Template

Templater Script

```javascript
<%*
const counterFile =
app.vault.getAbstractFileByPath(
"System/Counters.md"
);

const content =
await app.vault.read(counterFile);

const match =
content.match(
/incident_counter:\s*(\d+)/
);

const next =
parseInt(match[1]) + 1;

const padded =
String(next)
.padStart(4,"0");

const year =
moment().format("YYYY");

const title =
`INC-${year}-${padded}`;

await tp.file.rename(title);
%>
```

ผลลัพธ์

```text
INC-2026-0016
```

# 2. ฟังก์ชันใช้งานร่วมกัน

`Scripts/get-next-counter.js`

```javascript
module.exports = async (
  app,
  counterName
) => {

  const path =
    "System/Counters.md";

  const file =
    app.vault.getAbstractFileByPath(path);

  let content =
    await app.vault.read(file);

  const regex =
    new RegExp(
      `${counterName}:\\s*(\\d+)`
    );

  const match =
    content.match(regex);

  if (!match) {
    throw new Error(
      `Counter not found: ${counterName}`
    );
  }

  const next =
    parseInt(match[1]) + 1;

  content =
    content.replace(
      regex,
      `${counterName}: ${next}`
    );

  await app.vault.modify(
    file,
    content
  );

  return String(next)
    .padStart(4, "0");
};
```

# 2. QuickAdd Script

`Scripts/next-id.js`

```javascript
module.exports = async (params) => {

    const { app, quickAddApi } = params;

    const file =
        app.vault.getAbstractFileByPath(
            "System/Counters.md"
        );

    let content =
        await app.vault.read(file);

    const type =
        await quickAddApi.suggester(
            [
                "Incident",
                "RCA",
                "Change",
                "ERR-K8S",
                "ERR-DB",
                "ERR-CI"
            ],
            [
                "incident_counter",
                "rca_counter",
                "change_counter",
                "error_k8s_counter",
                "error_db_counter",
                "error_ci_counter"
            ]
        );

    const regex =
        new RegExp(
            `${type}:\\s*(\\d+)`
        );

    const match =
        content.match(regex);

    const next =
        Number(match[1]) + 1;

    content =
        content.replace(
            regex,
            `${type}: ${next}`
        );

    await app.vault.modify(
        file,
        content
    );

    const padded =
        String(next)
        .padStart(4, "0");

    quickAddApi.variables.id =
        padded;

    quickAddApi.variables.year =
        window.moment()
        .format("YYYY");
};
```

## เวอร์ชัน Safe Counter

`Scripts/next-id-safe.js`

```javascript
module.exports = async (params) => {

    const { app, quickAddApi } = params;

    const counterFile =
        app.vault.getAbstractFileByPath(
            "System/Counters.md"
        );

    let content =
        await app.vault.read(counterFile);

    const counterName =
        quickAddApi.variables.counterName;

    const regex =
        new RegExp(
            `${counterName}:\\s*(\\d+)`
        );

    const match =
        content.match(regex);

    if (!match)
        throw new Error(
            `Counter not found: ${counterName}`
        );

    let next =
        parseInt(match[1]) + 1;

    const prefix =
        quickAddApi.variables.prefix;

    const year =
        window.moment().format("YYYY");

    let filenameExists = true;

    while (filenameExists) {

        const padded =
            String(next)
            .padStart(4, "0");

        const id =
            `${prefix}-${year}-${padded}`;

        filenameExists =
            app.vault.getFiles()
            .some(
                f => f.basename.startsWith(id)
            );

        if (filenameExists)
            next++;
    }

    const padded =
        String(next)
        .padStart(4, "0");

    content =
        content.replace(
            regex,
            `${counterName}: ${next}`
        );

    await app.vault.modify(
        counterFile,
        content
    );

    quickAddApi.variables.id =
        padded;

    quickAddApi.variables.year =
        year;

    return true;
};
```

# Scripts/next-id-locked.js

```javascript
module.exports = async (params) => {

    const { app, quickAddApi } = params;

    const LOCK_PATH =
        "System/.counter-lock";

    const COUNTER_PATH =
        "System/Counters.md";

    const LOCK_TIMEOUT =
        30000; // 30 วินาที

    const counterName =
        quickAddApi.variables.counterName;

    const prefix =
        quickAddApi.variables.prefix;

    // =====================
    // Acquire Lock
    // =====================

    const existingLock =
        app.vault.getAbstractFileByPath(
            LOCK_PATH
        );

    if (existingLock) {

        const lockContent =
            await app.vault.read(
                existingLock
            );

        const lockTime =
            Number(lockContent);

        const age =
            Date.now() - lockTime;

        // lock ค้างเกิน timeout
        if (age > LOCK_TIMEOUT) {

            await app.vault.delete(
                existingLock
            );

        } else {

            throw new Error(
              `Counter locked by another process (${Math.floor(age/1000)} sec)`
            );
        }
    }

    await app.vault.create(
        LOCK_PATH,
        Date.now().toString()
    );

    try {

        // =====================
        // Read Counter
        // =====================

        const counterFile =
            app.vault
            .getAbstractFileByPath(
                COUNTER_PATH
            );

        let content =
            await app.vault.read(
                counterFile
            );

        const regex =
            new RegExp(
                `${counterName}:\\s*(\\d+)`
            );

        const match =
            content.match(regex);

        if (!match)
            throw new Error(
              `Counter not found: ${counterName}`
            );

        let next =
            parseInt(match[1]) + 1;

        const year =
            window.moment()
            .format("YYYY");

        // =====================
        // Duplicate Protection
        // =====================

        while (true) {

            const padded =
                String(next)
                .padStart(4, "0");

            const id =
                `${prefix}-${year}-${padded}`;

            const exists =
                app.vault
                .getFiles()
                .some(f =>
                    f.basename.startsWith(id)
                );

            if (!exists)
                break;

            next++;
        }

        const padded =
            String(next)
            .padStart(4, "0");

        // =====================
        // Update Counter
        // =====================

        content =
            content.replace(
                regex,
                `${counterName}: ${next}`
            );

        await app.vault.modify(
            counterFile,
            content
        );

        quickAddApi.variables.id =
            padded;

        quickAddApi.variables.year =
            year;

        quickAddApi.variables.fullId =
            `${prefix}-${year}-${padded}`;

    }
    finally {

        // =====================
        // Release Lock
        // =====================

        const lock =
            app.vault
            .getAbstractFileByPath(
                LOCK_PATH
            );

        if (lock) {

            await app.vault.delete(
                lock
            );
        }
    }
};
```

# Retry Version (Production)

หากต้องการให้รออัตโนมัติแทน Error

เพิ่มฟังก์ชัน

```javascript
async function sleep(ms){
  return new Promise(
      r => setTimeout(r, ms)
  );
}
```

แทนส่วน Lock

```javascript
const MAX_RETRY = 10;

let retry = 0;

while(
    app.vault.getAbstractFileByPath(
        LOCK_PATH
    )
){

    retry++;

    if(retry > MAX_RETRY){

        throw new Error(
          "Lock timeout exceeded"
        );
    }

    await sleep(1000);
}
```

ผลลัพธ์

```text
Device A -> Lock

Device B
  ↓
Waiting...
Waiting...
Waiting...
  ↓
Lock released
  ↓
Create next number
```

# ระดับ Enterprise ที่แนะนำ

ป้องกันเลขซ้ำ 4 ชั้น

```text
1. Counter Registry
       ↓
2. File Lock
       ↓
3. Lock Timeout
       ↓
4. Duplicate Filename Scan
```

Flow จริง:

```text
QuickAdd
    ↓
Acquire Lock
    ↓
Read Counter
    ↓
Check Existing Files
    ↓
Generate Next Number
    ↓
Update Counter
    ↓
Release Lock
    ↓
Create Note
```

รูปแบบนี้ค่อนข้างปลอดภัยสำหรับ Vault ขนาดใหญ่ที่ใช้ร่วมกับ Obsidian Sync, Git Sync หรือหลายอุปกรณ์พร้อมกัน และช่วยลดโอกาสเกิดเลขซ้ำได้มากที่สุดในสภาพแวดล้อมของ Obsidian.

แนะนำให้ยกระดับ Lock File จากไฟล์ที่เก็บแค่ timestamp เป็น **Lock Metadata File** ที่มีข้อมูลครบสำหรับ Debug และ Recovery

## Auto-Refresh Lock

สำหรับงานที่ใช้เวลานาน

```javascript
const refreshLock =
async () => {

    lockData.expiresAt =
        Date.now() + 30000;

    await app.vault.modify(
        lockFile,
        JSON.stringify(
            lockData,
            null,
            2
        )
    );
};
```

เรียกทุก 10 วินาที

```javascript
const heartbeat =
setInterval(
    refreshLock,
    10000
);
```

clearInterval(
