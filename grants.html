Dynamic Grant Resolution Engine

TypeScript / Node.js implementation for automated CRUD permission mapping and adaptive resolution.

TypeScript Engine
1. Definition of Grants & Operations
export enum Grant {
  READ = "READ",       // Viewing, searching, fetching
  CREATE = "CREATE",   // Adding, inserting, creating
  UPDATE = "UPDATE",   // Editing, updating, patching
  DELETE = "DELETE",   // Removing, destroying
  MANAGE = "MANAGE"    // Full administrative access
}

export interface OperationContext {
  action?: string;        // e.g., "getUser", "editProfile", "addNewDocument"
  method?: string;        // e.g., "GET", "POST", "PUT", "PATCH", "DELETE"
  path?: string;          // e.g., "/api/v1/reports"
  payload?: Record<string, any>;
  isOwner?: boolean;      // Ownership flags for context-aware auto-pick
}

export interface GrantResolution {
  requiredGrant: Grant;
  reason: string;
  confidence: "EXACT" | "INFERRED" | "DEFAULT";
}
2. Auto-Pick Engine Logic
export class GrantResolver {
  /**
   * Automatically picks the expected grant based on operation context.
   */
  public static autoPickGrant(context: OperationContext): GrantResolution {
    const { action, method, path, payload, isOwner } = context;

    // Rule 1: HTTP Method Mapping (Exact Match)
    if (method) {
      const verb = method.toUpperCase();
      switch (verb) {
        case "GET":
        case "HEAD":
        case "OPTIONS":
          return { requiredGrant: Grant.READ, reason: `HTTP Verb '${verb}' requires read access.`, confidence: "EXACT" };
        case "POST":
          return { requiredGrant: Grant.CREATE, reason: `HTTP Verb 'POST' creates a new resource.`, confidence: "EXACT" };
        case "PUT":
        case "PATCH":
          return { requiredGrant: Grant.UPDATE, reason: `HTTP Verb '${verb}' modifies an existing resource.`, confidence: "EXACT" };
        case "DELETE":
          return { requiredGrant: Grant.DELETE, reason: `HTTP Verb 'DELETE' removes a resource.`, confidence: "EXACT" };
      }
    }

    // Rule 2: Action Keyword Intent Recognition
    if (action) {
      const act = action.toLowerCase();

      // Read actions
      if (/(get|read|view|fetch|list|search|export|download|find)/.test(act)) {
        return { requiredGrant: Grant.READ, reason: `Action '${action}' detected as read-only.`, confidence: "INFERRED" };
      }

      // Add / Create actions
      if (/(add|create|insert|new|append|register|upload|publish)/.test(act)) {
        return { requiredGrant: Grant.CREATE, reason: `Action '${action}' detected as creation.`, confidence: "INFERRED" };
      }

      // Update / Edit actions
      if (/(edit|update|modify|patch|change|revise|toggle|set)/.test(act)) {
        return { requiredGrant: Grant.UPDATE, reason: `Action '${action}' detected as modification.`, confidence: "INFERRED" };
      }

      // Delete actions
      if (/(delete|remove|destroy|archive|trash|purge)/.test(act)) {
        return { requiredGrant: Grant.DELETE, reason: `Action '${action}' detected as deletion.`, confidence: "INFERRED" };
      }
    }

    // Rule 3: Payload Inspection Heuristic
    if (payload && Object.keys(payload).length > 0) {
      if (payload.id && Object.keys(payload).length > 1) {
        return { requiredGrant: Grant.UPDATE, reason: "Payload contains existing ID and updated fields.", confidence: "INFERRED" };
      }
      if (!payload.id) {
        return { requiredGrant: Grant.CREATE, reason: "Payload lacks ID; inferred new record insertion.", confidence: "INFERRED" };
      }
    }

    // Fallback: Default to Least Privilege (READ)
    return {
      requiredGrant: Grant.READ,
      reason: "Could not decisively infer mutation intent; defaulting to least privilege (READ).",
      confidence: "DEFAULT"
    };
  }

  /**
   * Verifies if user's granted permissions fulfill the required auto-picked grant.
   */
  public static evaluateAccess(userGrants: Grant[], context: OperationContext): { authorized: boolean; resolution: GrantResolution } {
    const resolution = this.autoPickGrant(context);

    // Full administrative grant bypasses specific checks
    if (userGrants.includes(Grant.MANAGE)) {
      return { authorized: true, resolution };
    }

    const authorized = userGrants.includes(resolution.requiredGrant);
    return { authorized, resolution };
  }
}

Key Logic Mechanisms
Feature Execution Strategy
HTTP Verb Parsing Maps standard HTTP verbs (GET → READ, POST → CREATE, PUT/PATCH → UPDATE, DELETE → DELETE).
Keyword Intent Matching Evaluates function or RPC action names against specific regex groups for read, add, update, and delete intents.
Payload Structural Inspection Distinguishes between CREATE and UPDATE operations based on the presence of existing entity identifiers (id).
Least-Privilege Fallback Defaults to READ when context is ambiguous to prevent accidental authorization escalation.
